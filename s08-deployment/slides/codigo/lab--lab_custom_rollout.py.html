1 """CP3 complemento REAL si agent/v1/responses no permite traffic splits. No sustituye Neptuno UI."""
2 import argparse,os,json
3 from pathlib import Path
4 import mlflow
5 from databricks.sdk import WorkspaceClient
6 from databricks.sdk.errors import NotFound
7 from databricks.sdk.service import serving
8 from mlflow.models.resources import DatabricksSQLWarehouse,DatabricksFunction
9 from lab_deploy import CONFIG,save
10 
11 def main():
12     p=argparse.ArgumentParser();p.add_argument('--profile');p.add_argument('--deploy',action='store_true');a=p.parse_args()
13     if a.profile:os.environ['DATABRICKS_CONFIG_PROFILE']=a.profile
14     w=WorkspaceClient();c=CONFIG
15     if a.deploy:
16         try: w.serving_endpoints.get(c['endpoint']+'-rollout')
17         except NotFound: pass
18         else: raise RuntimeError('Endpoint custom ya existe; usa lab_rollout.py, no recrear ni mover aliases')
19     mlflow.set_tracking_uri('databricks');mlflow.set_registry_uri('databricks-uc')
20     mlflow.set_experiment(f"/Users/{w.current_user.me().user_name}/AIS08-Neptuno")
21     name=f"{c['catalog']}.{c['schema']}.ventas_rollout";function=f"{c['catalog']}.{c['source_schema']}.ventas_categoria";versions=[]
22     import pandas as pd
23     for revision in ['champion','challenger']:
24         with mlflow.start_run(run_name='AIS08-custom-rollout-'+revision):
25             info=mlflow.pyfunc.log_model(name='ventas',python_model=str(Path(__file__).with_name('rollout_model.py')),
26                 model_config={'warehouse_id':c['warehouse_id'],'function_name':function,'revision':revision},
27                 resources=[DatabricksSQLWarehouse(warehouse_id=c['warehouse_id']),DatabricksFunction(function_name=function)],
28                 pip_requirements=['mlflow==3.16.1','databricks-sdk==0.143.0','pandas>=2,<4'],
29                 input_example=pd.DataFrame([{'categoria':'Bebidas','anio':2026}]))
30             version=mlflow.register_model(info.model_uri,name).version;versions.append(version)
31             mlflow.MlflowClient().set_registered_model_alias(name,revision,version)
32     endpoint=c['endpoint']+'-rollout'
33     save('custom-rollout-package',{'model':name,'versions':versions,'endpoint':endpoint,'source':'UC ventas S05 real','purpose':'Mecánica canary; ambas versiones preservan misma cifra y sólo etiqueta revision cambia'})
34     if a.deploy:
35         entities=[serving.ServedEntityInput(name=label,entity_name=name,entity_version=v,workload_size='Small',scale_to_zero_enabled=True) for label,v in zip(['champion','challenger'],versions)]
36         w.serving_endpoints.create(name=endpoint,config=serving.EndpointCoreConfigInput(name=endpoint,served_entities=entities,traffic_config=serving.TrafficConfig(routes=[serving.Route(served_model_name='champion',traffic_percentage=100),serving.Route(served_model_name='challenger',traffic_percentage=0)])))
37         print('Custom rollout endpoint submitted',endpoint)
38 if __name__=='__main__':main()
