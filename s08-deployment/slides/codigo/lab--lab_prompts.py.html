1 """CP7: prueba alias challenger y rollback; no altera el prompt de modelos ya servidos."""
2 import argparse,os
3 import mlflow
4 from lab_deploy import CONFIG,save
5 p=argparse.ArgumentParser();p.add_argument('--profile');a=p.parse_args()
6 if a.profile:os.environ['DATABRICKS_CONFIG_PROFILE']=a.profile
7 mlflow.set_registry_uri('databricks-uc')
8 name=f"{CONFIG['catalog']}.{CONFIG['schema']}.neptuno_system"
9 old=mlflow.genai.load_prompt(f'prompts:/{name}@champion')
10 candidate=mlflow.genai.register_prompt(name=name,template=old.template+'\nResponde con evidencia explícita.',commit_message='CP7 challenger didáctico; no promovido a Serving')
11 mlflow.genai.set_prompt_alias(name,'challenger',candidate.version)
12 # Simulación operacional REAL sobre el alias de prompt aislado; restaura siempre champion.
13 try:
14     mlflow.genai.set_prompt_alias(name,'champion',candidate.version)
15     promoted=mlflow.genai.load_prompt(f'prompts:/{name}@champion').version
16 finally:
17     mlflow.genai.set_prompt_alias(name,'champion',old.version)
18 restored=mlflow.genai.load_prompt(f'prompts:/{name}@champion').version
19 assert restored==old.version
20 save('prompts',{'name':name,'original':old.version,'challenger':candidate.version,'promoted_then_rolled_back':promoted,'restored':restored,'affects_existing_model':False})
21 print('Prompt alias rollback verified; endpoint template remains pinned')
