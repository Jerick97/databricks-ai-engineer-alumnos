1 -- Fixture explícito para probar extracción. No escribe inference tables ni acredita tráfico.
2 WITH fixtures(case_name,status_code,response) AS (
3   VALUES
4   ('valido',200,'{"output":[{"content":[{"type":"output_text","text":"Respuesta de prueba"}]}]}'),
5   ('texto_movido',200,'{"output":[{"text":"Respuesta en otra ruta"}]}'),
6   ('error_http',500,'{"error":"fallo de prueba"}')
7 )
8 SELECT case_name,
9  length(get_json_object(response,'$.output[0].content[0].text')) AS output_chars,
10  CASE WHEN status_code>=400 THEN 'http_error'
11       WHEN get_json_object(response,'$.output[0].content[0].text') IS NULL THEN 'parse_error'
12       ELSE 'ok' END AS parse_status
13 FROM fixtures ORDER BY case_name
