1 """CP4: crea App aislada y sube el código, sin tokens en archivos/browser."""
2 import os, json, argparse
3 from pathlib import Path
4 from databricks.sdk import WorkspaceClient
5 from databricks.sdk.errors import NotFound
6 from databricks.sdk.service import apps
7 from databricks.sdk.service.workspace import ImportFormat
8 from lab_deploy import CONFIG, save
9 
10 def main():
11     p=argparse.ArgumentParser();p.add_argument('--profile');a=p.parse_args()
12     if a.profile:os.environ['DATABRICKS_CONFIG_PROFILE']=a.profile
13     w=WorkspaceClient();c=CONFIG
14     assert c['app_name'].startswith('ais08-')
15     try: app=w.apps.get(c['app_name'])
16     except NotFound:
17         app=w.apps.create(app=apps.App(name=c['app_name'],description='S08 Neptuno: UI sobre Serving',resources=[apps.AppResource(name='neptuno-endpoint',serving_endpoint=apps.AppResourceServingEndpoint(name=c['endpoint'],permission=apps.AppResourceServingEndpointServingEndpointPermission.CAN_QUERY))])).result()
18     source=f"/Users/{w.current_user.me().user_name}/AIS08-labfiles/app"
19     w.workspace.mkdirs(source)
20     for path in (Path(__file__).resolve().parents[1]/'app').iterdir():
21         if path.is_file():w.workspace.upload(source+'/'+path.name,path.read_bytes(),format=ImportFormat.AUTO,overwrite=True)
22     deployment=w.apps.deploy(c['app_name'],app_deployment=apps.AppDeployment(source_code_path="/Workspace"+source))
23     save('app',{'name':app.name,'url':app.url,'source_code_path':source,'submitted':True})
24     print('App deployment submitted:',app.name)
25 if __name__=='__main__':main()
