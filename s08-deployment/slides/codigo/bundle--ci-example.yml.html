1 # Ejemplo GitHub Actions: copiar a .github/workflows sólo tras configurar identidad OIDC.
2 # No se ha ejecutado CI remoto; la validación local está en reports/lab-bundle-validation.json.
3 name: validate-neptuno
4 on: [pull_request, workflow_dispatch]
5 permissions:
6   contents: read
7   id-token: write
8 jobs:
9   validate:
10     runs-on: ubuntu-latest
11     environment: dev
12     env:
13       DATABRICKS_AUTH_TYPE: github-oidc
14       DATABRICKS_HOST: ${{ vars.DATABRICKS_HOST }}
15       DATABRICKS_CLIENT_ID: ${{ vars.DATABRICKS_CLIENT_ID }}
16       BUNDLE_VAR_catalog: ${{ vars.CATALOG }}
17       BUNDLE_VAR_schema: ${{ vars.SCHEMA }}
18       BUNDLE_VAR_app_name: ${{ vars.APP_NAME }}
19       BUNDLE_VAR_endpoint_name: ${{ vars.ENDPOINT_NAME }}
20       BUNDLE_VAR_warehouse_id: ${{ vars.WAREHOUSE_ID }}
21       BUNDLE_VAR_model_name: ${{ vars.MODEL_NAME }}
22       BUNDLE_VAR_model_version: ${{ vars.MODEL_VERSION }}
23     steps:
24       - uses: actions/checkout@v4
25       - uses: databricks/setup-cli@main
26       - run: databricks bundle validate -t dev
27         working-directory: s08-deployment/bundle
28       # Promoción: evaluaciones S06/S07 + smoke + aprobación environment prod.
29       # No desplegar automáticamente desde pull_request.
