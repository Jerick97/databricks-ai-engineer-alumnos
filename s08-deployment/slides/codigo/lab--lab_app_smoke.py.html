1 """Prueba HTTP oficial con auth unificada en servidor; jamás inyecta tokens en navegador."""
2 import argparse,os
3 import requests
4 from databricks.sdk import WorkspaceClient
5 from lab_deploy import CONFIG,save,REPORTS
6 p=argparse.ArgumentParser();p.add_argument('--profile');p.add_argument('--health-only',action='store_true');a=p.parse_args()
7 if a.profile:os.environ['DATABRICKS_CONFIG_PROFILE']=a.profile
8 w=WorkspaceClient();url=w.apps.get(CONFIG['app_name']).url.rstrip('/')
9 # Credenciales sólo en memoria, no imprimir headers ni Response.request.
10 headers=w.config.authenticate()
11 health=requests.get(url+'/health',headers=headers,timeout=30,allow_redirects=False)
12 result={'url':url,'health_status':health.status_code,'health_ok':health.status_code==200 and health.json().get('status')=='ok' if health.status_code==200 else False}
13 invalid=requests.post(url+'/',headers=headers,data={'question':'<script>alert(1)</script>'+'a'*2000},timeout=30,allow_redirects=False)
14 result.update(invalid_status=invalid.status_code,invalid_has_form='<form' in invalid.text,invalid_escaped='&lt;script&gt;' in invalid.text and '<script>alert(1)</script>' not in invalid.text)
15 if not a.health_only:
16     page=requests.post(url+'/',headers=headers,data={'question':'¿Cuánto vendimos de Bebidas en 2026?'},timeout=180,allow_redirects=False)
17     (REPORTS/'lab-app-observed.html').write_text(page.text)
18     result.update(post_status=page.status_code,has_expected_business_answer='116024.88' in page.text,has_source='gold.ventas_por_categoria_mes' in page.text)
19     result['pass']=result['health_ok'] and result['has_expected_business_answer'] and result['has_source'] and result['invalid_status']==400 and result['invalid_has_form'] and result['invalid_escaped']
20 else:result['pass']=result['health_ok']
21 save('app-http',result);print(result)
22 if not result['pass']:raise SystemExit(1)
