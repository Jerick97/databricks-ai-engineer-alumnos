1 """CP3: canary custom NeptunoVentas; agent/v1/responses no admite split."""
2 import argparse,json,os
3 from pathlib import Path
4 from mlflow import MlflowClient
5 import mlflow
6 from databricks.sdk import WorkspaceClient
7 from databricks.sdk.service import serving
8 from lab_deploy import CONFIG,save
9 
10 def normalized_snapshot(endpoint,workspace_host,model_name,config):
11     entities=sorted([serving.ServedEntityInput.from_dict(e.as_dict()).as_dict() for e in config.served_entities],key=lambda e:e['name'])
12     traffic=config.traffic_config.as_dict()
13     traffic['routes']=sorted(traffic['routes'],key=lambda r:r['served_model_name'])
14     return {'endpoint':endpoint,'workspace_host':workspace_host,'model_name':model_name,'served_entities':entities,'traffic_config':traffic}
15 
16 def validate_snapshot(snapshot,endpoint,workspace_host,model_name):
17     if any(snapshot.get(k)!=v for k,v in {'endpoint':endpoint,'workspace_host':workspace_host,'model_name':model_name}.items()):
18         raise ValueError('Snapshot pertenece a otro endpoint/workspace/modelo; no restaurar evidencia docente ni ajena')
19     if not snapshot.get('served_entities') or any(e.get('entity_name')!=model_name for e in snapshot['served_entities']):
20         raise ValueError('Snapshot contiene entidades de otro modelo')
21 
22 def main():
23     p=argparse.ArgumentParser();p.add_argument('--profile');p.add_argument('--challenger-version');p.add_argument('--percent',type=int,default=10);p.add_argument('--rollback',action='store_true');a=p.parse_args()
24     if a.profile:os.environ['DATABRICKS_CONFIG_PROFILE']=a.profile
25     if not 0<=a.percent<=100:raise ValueError('Porcentaje debe estar entre 0 y 100')
26     w=WorkspaceClient();c=dict(CONFIG)
27     c['endpoint']=CONFIG['endpoint']+'-rollout'
28     c['model_name']=f"{CONFIG['catalog']}.{CONFIG['schema']}.ventas_rollout"
29     mlflow.set_registry_uri('databricks-uc');client=MlflowClient()
30     current=w.serving_endpoints.get(c['endpoint'])
31     if current.task and current.task.startswith('agent/'):
32         raise ValueError('Traffic splitting no soportado para agentes; usa lab_custom_rollout.py')
33     if current.pending_config or current.state.ready.value!='READY':raise RuntimeError('Espera READY y NOT_UPDATING antes de otra actualización')
34     assert c['endpoint'].startswith('ais08-')
35     path=Path(__file__).resolve().parents[1]/'reports/lab-rollout-before.json'
36     if a.rollback:
37         old=json.loads(path.read_text())
38         validate_snapshot(old,c['endpoint'],w.config.host,c['model_name'])
39         w.serving_endpoints.update_config(c['endpoint'],served_entities=[serving.ServedEntityInput.from_dict(e) for e in old['served_entities']],traffic_config=serving.TrafficConfig.from_dict(old['traffic_config']))
40         print('Rollback solicitado; verificar READY y consulta antes de afirmar recuperación');return
41     if not a.challenger_version:raise ValueError('Especifica --challenger-version')
42     before=normalized_snapshot(c['endpoint'],w.config.host,c['model_name'],current.config)
43     if path.exists():
44         old=json.loads(path.read_text());validate_snapshot(old,c['endpoint'],w.config.host,c['model_name'])
45         if old!=before:raise ValueError('Ya existe snapshot de otro estado. Conservarlo; completar rollback o archivar explícitamente para nuevo ensayo')
46     else:save('rollout-before',before)
47     champion=client.get_model_version_by_alias(c['model_name'],'champion').version
48     client.set_registered_model_alias(c['model_name'],'challenger',a.challenger_version)
49     wanted={'champion':str(champion),'challenger':str(a.challenger_version)}
50     existing={e.name:e for e in current.config.served_entities}
51     # Preservar parámetros exactos si las versiones ya están desplegadas; no cambiar workload/env.
52     entities=[serving.ServedEntityInput.from_dict(existing[name].as_dict()) if name in existing and existing[name].entity_version==version else serving.ServedEntityInput(name=name,entity_name=c['model_name'],entity_version=version,workload_size='Small',scale_to_zero_enabled=True) for name,version in wanted.items()]
53     routes=[serving.Route(served_model_name='champion',traffic_percentage=100-a.percent),serving.Route(served_model_name='challenger',traffic_percentage=a.percent)]
54     # REST exige entidades incluso para cambio de rutas (lab-rollout-api-limitation.json).
55     w.serving_endpoints.update_config(c['endpoint'],served_entities=entities,traffic_config=serving.TrafficConfig(routes=routes))
56     save('rollout',{'endpoint':c['endpoint'],'champion':champion,'challenger':a.challenger_version,'percent':a.percent,'submitted':True})
57 if __name__=='__main__':main()
