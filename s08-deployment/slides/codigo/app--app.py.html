1 """UI servidor: el navegador nunca recibe credenciales Databricks."""
2 import os
3 import uuid
4 from flask import Flask, request, render_template_string
5 from databricks.sdk import WorkspaceClient
6 app=Flask(__name__)
7 app.config['MAX_CONTENT_LENGTH']=12000
8 PAGE='''<!doctype html><html lang="es"><meta charset="utf-8"><meta name="viewport" content="width=device-width"><title>Neptuno · S08</title>
9 <style>body{font:18px system-ui;max-width:820px;margin:50px auto;padding:24px;color:#123}textarea{box-sizing:border-box;width:100%;height:110px;font:inherit}button{font:inherit;padding:12px;background:#e44332;color:white;border:0}pre{white-space:pre-wrap;overflow-wrap:anywhere;line-height:1.6;background:#f2f5f7;padding:20px}</style>
10 <h1>Copiloto de Datos Neptuno</h1><p>Consulta ventas por categoría y año, reposición o políticas documentadas.</p>
11 <form method="post"><label for="q">Pregunta</label><textarea id="q" name="question" maxlength="2000" required>{{ question }}</textarea><p><button>Consultar</button></p></form>
12 {% if answer %}<h2>Respuesta</h2><pre>{{ answer }}</pre>{% endif %}
13 <p>Acceso corporativo mediante Databricks Apps. El servidor consulta con la identidad de la App, no con los permisos individuales del usuario. No ingreses datos personales ni secretos.</p></html>'''
14 @app.route('/',methods=['GET','POST'])
15 def index():
16     answer='';question=''
17     if request.method=='POST':
18         question=request.form.get('question','').strip()
19         if not question or len(question)>2000:
20             return render_template_string(PAGE,question=question[:2000],answer='Pregunta vacía o demasiado larga. Escribe entre 1 y 2000 caracteres y vuelve a intentar.'),400
21         try:
22             w=WorkspaceClient()
23             result=w.api_client.do('POST',f"/serving-endpoints/{os.environ['SERVING_ENDPOINT']}/invocations",body={'input':[{'role':'user','content':question}], 'client_request_id':str(uuid.uuid4())})
24             answer='\n'.join(part.get('text','') for item in result.get('output',[]) for part in item.get('content',[]) if part.get('type')=='output_text') or 'Respuesta sin texto; revisa el endpoint.'
25         except Exception:
26             answer='La consulta no pudo completarse. Revisa el estado del endpoint con el instructor.'
27     return render_template_string(PAGE,question=question,answer=answer)
28 @app.get('/health')
29 def health():return {'status':'ok'}
