18 PATTERNS = (r'ignora (todas |las )?(instrucciones|reglas)', r'ignore (all |previous )?instructions',
19             r'(revela|muestra|imprime).{0,30}(token|secreto|system prompt)', r'drop\s+table')
20 SECRET = re.compile(r'\bdapi[a-zA-Z0-9]{20,}\b|\bBearer\s+[a-zA-Z0-9._-]{20,}', re.I)
21 EMAIL = re.compile(r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}\b')

23 def check_text(text, limit=2000):
24     if not isinstance(text, str) or not text.strip():
25         raise Blocked('entrada_vacia')
26     if len(text) > limit:
27         raise Blocked('limite_caracteres')
28     if any(re.search(p, text, re.I) for p in PATTERNS):
29         raise Blocked('patron_inyeccion_conocido')
30     if SECRET.search(text):
31         raise Blocked('posible_credencial')
32     return text

34 def moderate(text, classifier):
35     """classifier devuelve el formato nativo safe / unsafe + categorías de Llama Guard."""
36     try:
37         result = classifier(text).strip()
38     except Exception as exc:
39         raise Blocked('moderacion_no_disponible') from exc
40     lines = result.splitlines()
41     if not lines or lines[0].strip().lower() not in {'safe', 'unsafe'}:
42         raise Blocked('moderacion_formato_invalido')
43     if lines[0].strip().lower() == 'safe' and len(lines) != 1:
44         raise Blocked('moderacion_formato_invalido')
45     if lines[0].strip().lower() == 'unsafe':
46         raise Blocked('moderacion_unsafe')
47     return {'verdict': 'safe'}

49 def sanitize_output(text):
50     if SECRET.search(text):
51         raise Blocked('salida_con_credencial')
52     return EMAIL.sub('[EMAIL OCULTO]', text)

55 def gateway_verdict(invoke, text):
56     # The gateway, not the generative Llama Instruct model, judges the submitted text.
57     try:
58         # Submit the text unchanged: an added benign system instruction changed
59         # the observed safety verdict in the first remote integration test.
60         response = invoke({'messages': [{'role': 'user', 'content': text}],
61                            'max_tokens': 4, 'temperature': 0})
62     except Exception as exc:
63         try:
64             error = json.loads(str(exc))
65         except (ValueError, TypeError):
66             raise RuntimeError('gateway_unavailable') from exc
67         # Only the explicit guardrail error is a safety rejection; auth/timeouts remain errors.
68         if error.get('finishReason') in {'input_guardrail_triggered', 'output_guardrail_triggered'}:
69             field = 'input_guardrail' if error['finishReason'].startswith('input') else 'output_guardrail'
70             if any(item.get('flagged') is True for item in error.get(field, [])):
71                 return 'unsafe'
72         raise RuntimeError('gateway_unavailable') from exc
73     choices = response.get('choices') if isinstance(response, dict) else None
74     if not isinstance(choices, list) or not choices:
75         raise RuntimeError('gateway_invalid_response')
76     for choice in choices:
77         if not isinstance(choice, dict) or choice.get('finish_reason') not in {'stop', 'length'}:
78             raise RuntimeError('gateway_invalid_response')
79         message = choice.get('message')
80         if not isinstance(message, dict) or message.get('role') != 'assistant' or not isinstance(message.get('content'), str) or not message['content'].strip():
81             raise RuntimeError('gateway_invalid_response')
82     return 'safe'

104 def validate_call(name, args):
105     definitions={t['function']['name']:t['function']['parameters'] for t in TOOLS}
106     if name not in definitions:
107         raise ValueError('Herramienta fuera de la allowlist')
108     validate(args, definitions[name])

110 class AgenteNeptuno(ResponsesAgent):
111     def __init__(self, config=None):
112         self.config = config or CONFIG
113         for key in ['catalog','schema','source_schema','approved_schema']:
114             if not re.fullmatch(r'[a-zA-Z_][a-zA-Z0-9_]*',self.config[key]):
115                 raise ValueError('Identificador UC inválido')
116         self.w = WorkspaceClient()
117         self.llm = self.w.serving_endpoints.get_open_ai_client()
118         self.prompt = self.config.get('system_prompt',SYSTEM)
119 
120     def sql(self, statement, parameters=None):
121         r = self.w.statement_execution.execute_statement(warehouse_id=self.config['warehouse_id'],statement=statement,
122              parameters=[StatementParameterListItem(name=k,value=str(v),type='INT' if isinstance(v,int) else 'STRING') for k,v in (parameters or {}).items()], wait_timeout='50s')
123         deadline=time.monotonic()+120
124         while r.status.state.value in {'PENDING','RUNNING'} and time.monotonic()<deadline:
125             time.sleep(2)
126             r=self.w.statement_execution.get_statement(r.statement_id)
127         if r.status.state.value!='SUCCEEDED':
128             raise RuntimeError('Consulta no completada: '+str(r.status.state.value))
129         if r.manifest and r.manifest.truncated:
130             raise ValueError('Resultado SQL truncado; reduce el conjunto')
131         names=[c.name for c in r.manifest.schema.columns]
132         return [dict(zip(names,row)) for row in (r.result.data_array or [])]

134     @mlflow.trace(span_type='TOOL')
135     def tool(self, name, args):
136         validate_call(name,args)
137         c=self.config['catalog']; s=self.config['source_schema']
138         if name in {'ventas_categoria','productos_reponer'}:
139             params=':p_categoria, :p_anio' if name=='ventas_categoria' else ''
140             return json.loads(self.sql(f'SELECT {c}.{s}.{name}({params}) AS result',args)[0]['result'])
141         rows=self.sql(f'SELECT chunk_id, documento_id, titulo, texto_citable, to_json(embedding) AS embedding FROM {c}.rag.chunks_embeddings LIMIT 1001')
142         if not rows or len(rows)>1000: raise ValueError('Corpus ausente o demasiado grande; usa Vector Search')
143         query=self.llm.embeddings.create(model=self.config['embedding_endpoint'],input=args['pregunta']).data[0].embedding
144         def similarity(row):
145             vector=json.loads(row['embedding'])
146             if len(vector)!=len(query): raise ValueError('Dimensión de embeddings incompatible')
147             norm=math.sqrt(sum(x*x for x in vector)*sum(x*x for x in query))
148             return sum(a*b for a,b in zip(vector,query))/norm if norm else 0
149         ranked=sorted([(similarity(r),r) for r in rows],key=lambda item:item[0],reverse=True)[:3]
150         return {'fuente':f'{c}.rag.chunks_embeddings','documentos':[{'score':score,**{k:v for k,v in r.items() if k!='embedding'}} for score,r in ranked]}

152     @mlflow.trace(span_type='AGENT')
153     def predict(self, request: ResponsesAgentRequest) -> ResponsesAgentResponse:
154         started=time.monotonic(); audit=[]; tokens_in=0; tokens_out=0
155         messages=[{'role':'system','content':self.prompt}]
156         evidence=[]
157         for item in request.input:
158             d=item.model_dump(exclude_none=True) if hasattr(item,'model_dump') else item
159             if d.get('role') not in {'user','assistant'}: raise ValueError('Solo user/assistant')
160             content=d.get('content','')
161             if isinstance(content,list): content='\n'.join(x.get('text','') for x in content)
162             if not isinstance(content,str) or len(content)>8000: raise ValueError('Texto inválido o >8000 caracteres')
163             check_text(content)
164             self.safety(content)
165             messages.append({'role':d['role'],'content':content})
166         user_text=' '.join(m['content'] for m in messages if m['role']=='user')
167         if re.search(r'margen|rentabilidad|costos',user_text,re.I):
168             return self.response('No puedo calcular margen ni rentabilidad: faltan datos de costos. Puedo consultar venta neta.',audit,started,0,0)
169         if re.search(r'vendi|ventas?',user_text,re.I) and not re.search(r'\b(?:19|20)\d{2}\b',user_text):
170             return self.response('¿De qué año necesitas la venta neta? Indica también la categoría.',audit,started,0,0)
171         answer='Se alcanzó el límite; reformula la pregunta.'
172         for _ in range(self.config.get('max_turns',6)):
173             completion=self.llm.chat.completions.create(model=self.config['llm_endpoint'],messages=messages,tools=TOOLS,temperature=0,max_tokens=1000)
174             if completion.usage:
175                 tokens_in+=completion.usage.prompt_tokens or 0; tokens_out+=completion.usage.completion_tokens or 0
176             msg=completion.choices[0].message
177             if not msg.tool_calls:
178                 answer=msg.content or 'Sin respuesta'
179                 if isinstance(answer,list): answer='\n'.join(p.get('text','') for p in answer if p.get('type') in {'text','output_text'})
180                 break
181             payload={'role':'assistant','content':None,'tool_calls':[call.model_dump(exclude_none=True) for call in msg.tool_calls]}
182             messages.append(payload)
183             for call in msg.tool_calls:
184                 if len(audit)>=self.config.get('max_tool_calls',4): return self.response(answer,audit,started,tokens_in,tokens_out)
185                 try:
186                     args=json.loads(call.function.arguments)
187                     if call.function.name=='buscar_documentos': args['pregunta']=user_text[:500]
188                     if call.function.name=='ventas_categoria' and (str(args.get('p_anio')) not in user_text or str(args.get('p_categoria','')).casefold() not in user_text.casefold()):
189                         raise ValueError('Año o categoría no especificados por usuario; solicita aclaración')
190                     result={'ok':True,'data':self.tool(call.function.name,args)}
191                 except Exception as exc:
192                     result={'ok':False,'error':type(exc).__name__}
193                 try:
194                     check_text(json.dumps(result,ensure_ascii=False),limit=16000)
195                 except Blocked:
196                     result={'ok':False,'error':'contenido_tool_bloqueado'}
197                 if result['ok']: evidence.append((call.function.name,result['data']))
198                 audit.append({'tool':call.function.name,'ok':result['ok']})
199                 messages.append({'role':'tool','tool_call_id':call.id,'content':json.dumps(result,ensure_ascii=False)[:16000]})
200         if evidence:
201             answer=self.render_evidence(evidence)
202         return self.response(answer,audit,started,tokens_in,tokens_out)

204     def render_evidence(self, evidence):
205         # La selección de herramientas es del LLM; cifras y citas se renderizan del resultado validado.
206         parts=[]
207         for name,data in evidence:
208             if name=='ventas_categoria':
209                 if not data.get('filas') or data.get('venta_neta') is None:
210                     parts.append(f"No hay ventas para {data['categoria']} en {data['anio']}. Fuente: {data['fuente']}.")
211                 else:
212                     parts.append(f"Venta neta de {data['categoria']} en {data['anio']}: {data['venta_neta']}. Fuente: {data['fuente']}.")
213             elif name=='productos_reponer':
214                 products=data.get('productos',[])
215                 parts.append('Productos que requieren reposición según la tabla de inventario:\n'+('\n'.join(f"{p['producto']} (stock {p['stock']}, en camino {p['en_camino']}, punto de reorden {p['punto_reorden']})" for p in products) if products else 'ningún producto')+f". Fuente: {data['fuente']}.")
216             else:
217                 parts.append('Evidencia recuperada (fragmentos; no una política completa):\n'+'\n'.join(f"{d['texto_citable']} [documento_id={d['documento_id']}; chunk_id={d['chunk_id']}]" for d in data['documentos']))
218         return '\n\n'.join(parts)

220     def safety(self, text):
221         endpoint=self.config['moderation_endpoint']
222         verdict=gateway_verdict(lambda body:self.w.api_client.do('POST',f'/serving-endpoints/{endpoint}/invocations',body=body),text)
223         if verdict!='safe': raise Blocked('moderacion_unsafe')
224 
225     def response(self, text, audit, started, tokens_in, tokens_out):
226         self.safety(text)
227         return ResponsesAgentResponse(output=[self.create_text_output_item(text=sanitize_output(text),id=str(uuid.uuid4()))],custom_outputs={
228             'tool_audit':audit,'tool_calls':len(audit),'tool_errors':sum(not a['ok'] for a in audit),
229             'latency_ms':round((time.monotonic()-started)*1000),'input_tokens':tokens_in,'output_tokens':tokens_out,
230             'moderation':'ai_gateway_input_output','prompt_version':self.config.get('prompt_version','embedded'),'agent_revision':'s08-v2','llm_endpoint':self.config['llm_endpoint']})
