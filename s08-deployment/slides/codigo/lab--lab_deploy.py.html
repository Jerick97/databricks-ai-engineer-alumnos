1 """CP0–CP2: despliegue real, aislado. Ejecutar desde lab con auth unificada."""
2 import argparse
3 import json
4 import os
5 from pathlib import Path
6 import mlflow
7 from mlflow import MlflowClient
8 from mlflow.models.resources import DatabricksServingEndpoint, DatabricksSQLWarehouse, DatabricksFunction, DatabricksTable
9 from databricks.sdk import WorkspaceClient
10 from databricks.sdk.errors import NotFound
11 from databricks.sdk.service import serving
12 
13 ROOT=Path(__file__).resolve().parent
14 CONFIG=json.loads((ROOT/'config.json').read_text())
15 REPORTS=ROOT.parent/'reports'
16 
17 def save(name,data):
18     REPORTS.mkdir(exist_ok=True)
19     (REPORTS/f'lab-{name}.json').write_text(json.dumps(data,indent=2,default=str,ensure_ascii=False))
20 
21 def main():
22     p=argparse.ArgumentParser();p.add_argument('--profile');p.add_argument('--package-only',action='store_true');args=p.parse_args()
23     if args.profile: os.environ['DATABRICKS_CONFIG_PROFILE']=args.profile
24     w=WorkspaceClient(); c=CONFIG
25     assert c['schema'].startswith('ais08') and c['endpoint'].startswith('ais08-')
26     try:w.schemas.get(f"{c['catalog']}.{c['schema']}")
27     except NotFound:w.schemas.create(name=c['schema'],catalog_name=c['catalog'],comment='Laboratorio aislado S08')
28     mlflow.set_tracking_uri('databricks');mlflow.set_registry_uri('databricks-uc')
29     mlflow.set_experiment(f"/Users/{w.current_user.me().user_name}/AIS08-Neptuno")
30     from agent import SYSTEM
31     prompt=mlflow.genai.register_prompt(name=f"{c['catalog']}.{c['schema']}.neptuno_system",template=SYSTEM,commit_message='S08 governed deployment')
32     try: mlflow.genai.load_prompt(f'prompts:/{prompt.name}@champion')
33     except Exception: mlflow.genai.set_prompt_alias(prompt.name,'champion',prompt.version)
34     c['system_prompt']=prompt.template
35     c['prompt_version']=str(prompt.version)
36     resources=[DatabricksServingEndpoint(endpoint_name=c['llm_endpoint']),DatabricksServingEndpoint(endpoint_name=c['embedding_endpoint']),DatabricksSQLWarehouse(warehouse_id=c['warehouse_id']),DatabricksTable(table_name=f"{c['catalog']}.rag.chunks_embeddings")]
37     resources += [DatabricksServingEndpoint(endpoint_name=c['moderation_endpoint'])]
38     resources += [DatabricksFunction(function_name=f"{c['catalog']}.{c['source_schema']}.{f}") for f in ['ventas_categoria','productos_reponer']]
39     with mlflow.start_run(run_name='AIS08-agent-as-code'):
40         info=mlflow.pyfunc.log_model(name='agent',python_model=str(ROOT/'agent.py'),model_config=c,resources=resources,
41             pip_requirements=['mlflow=='+mlflow.__version__,'databricks-sdk==0.143.0','openai==2.54.0','jsonschema==4.26.0'],
42             input_example={'input':[{'role':'user','content':'¿Cuánto vendimos de Bebidas en 2026?'}]})
43         model=mlflow.register_model(info.model_uri,c['model_name'])
44     client=MlflowClient()
45     try: client.get_model_version_by_alias(c['model_name'],'champion')
46     except Exception: client.set_registered_model_alias(c['model_name'],'champion',model.version)
47     save('package',{'model_name':c['model_name'],'version':model.version,'model_uri':info.model_uri,'mlflow_version':mlflow.__version__})
48     if args.package_only:return
49     # Un alias del Registry NO cambia el tráfico: Serving fija una versión numérica.
50     entity=serving.ServedEntityInput(name='champion',entity_name=c['model_name'],entity_version=str(model.version),workload_size='Small',scale_to_zero_enabled=True)
51     try:w.serving_endpoints.get(c['endpoint'])
52     except NotFound:
53         result=w.serving_endpoints.create(name=c['endpoint'],config=serving.EndpointCoreConfigInput(name=c['endpoint'],served_entities=[entity]),
54             ai_gateway=serving.AiGatewayConfig(inference_table_config=serving.AiGatewayInferenceTableConfig(enabled=True,catalog_name=c['catalog'],schema_name=c['schema'],table_name_prefix='ais08_neptuno')))
55         save('deployment',{'endpoint':c['endpoint'],'submitted':True,'version':model.version})
56     else:
57         raise RuntimeError('Endpoint ya existe: inspecciona y usa lab_rollout.py; no reemplazar silenciosamente')
58 
59 if __name__=='__main__':main()
